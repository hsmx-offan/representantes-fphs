const app = initializeApp(firebaseConfig); const auth = getAuth(app); const cargando = document.getElementById("cargando"); const contenido = document.getElementById("contenido"); onAuthStateChanged(auth, (user) => { if (user) { cargando.style.display = "none"; contenido.style.display = "block"; } else { window.location.href = "./"; } }); window.logoutAdmin = async function () { await signOut(auth); window.location.href = "./"; };
